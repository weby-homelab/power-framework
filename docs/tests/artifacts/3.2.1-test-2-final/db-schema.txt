# Database Schema — TEST-2 (WS)

## Core Tables

### fts_notes (FTS5 Virtual Table)
```sql
CREATE VIRTUAL TABLE fts_notes USING fts5(
    title, tags, description, content, rel_path, note_type,
    tokenize = 'unicode61 remove_diacritics 2'
);
```

### file_metadata
```sql
CREATE TABLE file_metadata (
    rel_path    TEXT PRIMARY KEY,
    mtime       REAL NOT NULL
);
```

### tf_vectors
```sql
CREATE TABLE tf_vectors (
    rel_path    TEXT PRIMARY KEY,
    tf_data     TEXT NOT NULL,  -- JSON dict: term -> tf score
    mtime       REAL NOT NULL
);
```

### doc_embeddings
```sql
CREATE TABLE doc_embeddings (
    rel_path    TEXT PRIMARY KEY,
    embedding   BLOB NOT NULL,   -- 1024 * 4 = 4096 bytes (float32)
    mtime       REAL NOT NULL
);
```

### chunk_embeddings
```sql
CREATE TABLE chunk_embeddings (
    chunk_id    TEXT PRIMARY KEY,  -- "rel_path::chunk_N"
    rel_path    TEXT NOT NULL,
    embedding   BLOB NOT NULL,     -- 1024 * 4 = 4096 bytes
    content     TEXT NOT NULL,     -- chunk text for snippets
    mtime       REAL NOT NULL
);
CREATE INDEX idx_chunk_embeddings_rel_path ON chunk_embeddings(rel_path);
```

### dense_index_manifest
```sql
CREATE TABLE dense_index_manifest (
    manifest_key   TEXT PRIMARY KEY,
    manifest_value TEXT NOT NULL
);
-- Expected keys: schema_version, embedding_dimension, chunk_count, embedding_provider, embedding_model
```

### relations
```sql
CREATE TABLE relations (
    subject   TEXT NOT NULL,
    predicate TEXT NOT NULL,
    object    TEXT NOT NULL,
    source    TEXT NOT NULL,  -- rel_path of source note
    PRIMARY KEY (subject, predicate, object)
);
CREATE INDEX idx_relations_subject ON relations(subject);
CREATE INDEX idx_relations_object ON relations(object);
```
